namespace at {
namespace detail {

inline Tensor tensor_md_2d_uint8(const std::vector<std::vector<uint8_t>>& values) {
    int64_t d0 = (int64_t)values.size();
    int64_t d1 = values.empty() ? 0 : (int64_t)values[0].size();
    std::vector<uint8_t> flat;
    flat.reserve(d0 * d1);
    for (const auto& r : values) for (auto v : r) flat.push_back(v);
    return at::tensor(flat, TensorOptions().dtype(at::kByte)).reshape({d0, d1});
}
